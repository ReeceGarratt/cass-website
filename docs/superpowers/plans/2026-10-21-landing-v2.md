```js
// Ink bounding box of a region: pixels that differ from the region's
// top-left colour. Usage: node measure.mjs <png> <x> <y> <w> <h> [crop.png]
import { createRequire } from 'node:module';
const sharp = createRequire('F:/Code/cass-website-landing/package.json')(
  'sharp',
);
const [src, x, y, w, h, out] = process.argv.slice(2);
const r = { left: +x, top: +y, width: +w, height: +h };
const img = sharp(src).extract(r);
if (out) await img.clone().toFile(out);
const { data, info } = await img
  .ensureAlpha()
  .raw()
  .toBuffer({ resolveWithObject: true });
const bg = [data[0], data[1], data[2]];
let [minX, minY, maxX, maxY] = [Infinity, Infinity, -1, -1];
for (let j = 0; j < info.height; j++)
  for (let i = 0; i < info.width; i++) {
    const k = (j * info.width + i) * 4;
    const d =
      Math.abs(data[k] - bg[0]) +
      Math.abs(data[k + 1] - bg[1]) +
      Math.abs(data[k + 2] - bg[2]);
    if (d > 60) {
      minX = Math.min(minX, i);
      maxX = Math.max(maxX, i);
      minY = Math.min(minY, j);
      maxY = Math.max(maxY, j);
    }
  }
console.log(
  `ink x ${minX + r.left}..${maxX + r.left} y ${minY + r.top}..${maxY + r.top}`,
);
```

`$T/pair.mjs`:

```js
// The same region from the Figma screenshot (top) and a page shot
// (bottom), stacked into one PNG, plus the share of differing pixels.
// Usage: node pair.mjs <figma.png> <shot.png> <x> <y> <w> <h> <out.png>
import { createRequire } from 'node:module';
const sharp = createRequire('F:/Code/cass-website-landing/package.json')(
  'sharp',
);
const [a, b, x, y, w, h, out] = process.argv.slice(2);
const r = { left: +x, top: +y, width: +w, height: +h };
const [A, B] = await Promise.all(
  [a, b].map((f) => sharp(f).extract(r).removeAlpha().png().toBuffer()),
);
await sharp({
  create: {
    width: r.width,
    height: r.height * 2 + 4,
    channels: 3,
    background: '#ff00ff',
  },
})
  .composite([
    { input: A, top: 0, left: 0 },
    { input: B, top: r.height + 4, left: 0 },
  ])
  .png()
  .toFile(out);
const [RA, RB] = await Promise.all(
  [A, B].map((buf) => sharp(buf).raw().toBuffer()),
);
let d = 0;
for (let i = 0; i < RA.length; i += 3)
  if (
    Math.abs(RA[i] - RB[i]) +
      Math.abs(RA[i + 1] - RB[i + 1]) +
      Math.abs(RA[i + 2] - RB[i + 2]) >
    60
  )
    d++;
console.log(
  `${out}: ${((100 * d) / (r.width * r.height)).toFixed(2)}% of pixels differ`,
);
```

- [ ] **Step 4: Write the shared copy script**

`src/scripts/copy-email.ts` is phase 2's script from `SiteNav`, generalised from one link to all of them:

```ts
// Reason for client JS (invariant 3), D-038: "Email me" and the landing
// page's email pills copy the address to the clipboard, which has no
// HTML-only equivalent. Without JS, or if copying fails, each link opens
// its mailto: as normal.
//
// The tick is the landing page's (272:2302), used everywhere (D-048). How
// long it shows is undesigned (Reece approved 3 seconds).
const COPIED_MS = 3000;
// One live region per page. SiteNav renders it, and every page has the nav.
const liveRegion = document.querySelector<HTMLElement>('[data-copy-status]');

for (const link of document.querySelectorAll<HTMLAnchorElement>(
  '[data-copy-email]',
)) {
  let timer = 0;
  link.addEventListener('click', async (event) => {
    // Leave modified clicks (new tab, new window) to the browser.
    if (event.metaKey || event.ctrlKey || event.shiftKey || event.altKey)
      return;
    event.preventDefault();
    try {
      // Read at click time: navigator.clipboard is missing outside
      // secure contexts, and the TypeError lands in the catch.
      await navigator.clipboard.writeText(link.dataset.copyEmail ?? '');
    } catch {
      location.href = link.href;
      return;
    }
    link.toggleAttribute('data-copied', true);
    if (liveRegion) {
      // Clear first, so a second copy is announced again.
      liveRegion.textContent = '';
      requestAnimationFrame(() => {
        liveRegion.textContent = 'Email address copied';
      });
    }
    clearTimeout(timer);
    timer = window.setTimeout(() => {
      link.removeAttribute('data-copied');
      if (liveRegion) liveRegion.textContent = '';
    }, COPIED_MS);
  });
}

// A module, not a global script, so its names can't clash with the DOM's
// globals.
export {};
```
